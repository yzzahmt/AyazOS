cmake_minimum_required(VERSION 3.21)
project(ayaz-app VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Gui Qml Quick)
add_executable(ayaz-app main.cpp native.cpp native.h)
target_link_libraries(ayaz-app PRIVATE Qt6::Gui Qt6::Qml Qt6::Quick)
install(TARGETS ayaz-app RUNTIME DESTINATION bin)
